const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const path = require('path'); const app = express(); const server = http.createServer(app); const io = new Server(server); app.use(express.json()); app.use(express.static(path.join(__dirname, 'public'))); // سلم رتب LSPD مع الأوزان للفرز (الرتب الأعلى تظهر أولاً) const RANK_WEIGHTS = { 'Chief of Police': 110, 'Assistant Chief': 100, 'Deputy Chief': 90, 'Captain': 80, 'Lieutenant': 70, 'Staff Sergeant': 60, 'First Sergeant': 50, 'Sergeant': 40, 'Senior Lead Officer': 30, 'Officer III': 25, 'Officer II': 20, 'Officer I': 15, 'Cadet': 10 }; // بيانات الوينقات المتاحة const availableWings = [ { id: 'swat', name: 'قوات SWAT', icon: 'fa-user-ninja' }, { id: 'air', name: 'الطيران العمودي (Air Unit)', icon: 'fa-helicopter' }, { id: 'k9', name: 'وحدة K-9', icon: 'fa-dog' }, { id: 'traffic', name: 'المرور والسير', icon: 'fa-car' }, { id: 'investigation', name: 'التحقيقات CID', icon: 'fa-magnifying-glass' }, { id: 'academy', name: 'مدرب الأكاديمية', icon: 'fa-graduation-cap' } ]; // قاعدة بيانات مؤقتة في الذاكرة (Memory DB) let cadets = [ { discordId: '1001', name: 'Ebrahem Derbah', rank: 'First Sergeant', hours: 45.5, points: 120, wings: ['swat', 'air', 'academy'], reports: [{ title: 'تقرير دورية', content: 'تم ضبط مخالفات في السرعة', date: '2026-08-22' }] }, { discordId: '1002', name: 'John Doe', rank: 'Cadet', hours: 12.0, points: 30, wings: ['traffic'], reports: [] } ]; let users = []; let logs = []; // دالة فرز العسكريين حسب وزن الرتبة function sortCadets(data) { return data.sort((a, b) => { const weightA = RANK_WEIGHTS[a.rank] || 0; const weightB = RANK_WEIGHTS[b.rank] || 0; return weightB - weightA; }); } // REST APIs app.get('/api/wings-list', (req, res) => { res.json(availableWings); }); app.get('/api/cadets', (req, res) => { res.json(sortCadets(cadets)); }); app.get('/api/users', (req, res) => { res.json(users); }); app.get('/api/logs', (req, res) => { res.json(logs); }); app.post('/api/login-request', (req, res) => { const { username, copyId } = req.body; let user = users.find(u => u.copyId === copyId); if (!user) { user = { username, copyId, approved: false, role: 'viewer', status: 'active', lastSeen: Date.now() }; users.push(user); } else { user.username = username; user.lastSeen = Date.now(); } io.emit('usersUpdate', users); res.json({ approved: user.approved, role: user.role }); }); app.post('/api/user-heartbeat', (req, res) => { const { copyId, status } = req.body; const user = users.find(u => u.copyId === copyId); if (user) { user.status = status; user.lastSeen = Date.now(); io.emit('usersUpdate', users); } res.sendStatus(200); }); app.post('/api/approve-user', (req, res) => { const { copyId, approve, adminName } = req.body; const userIndex = users.findIndex(u => u.copyId === copyId); if (userIndex !== -1) { const targetUser = users[userIndex]; if (approve) { targetUser.approved = true; logs.unshift({ by: adminName || 'Admin', action: 'موافقة على دخول المستخدم', target: targetUser.username, time: new Date().toLocaleString('ar-SA') }); } else { users.splice(userIndex, 1); logs.unshift({ by: adminName || 'Admin', action: 'رفض طلب دخول', target: targetUser.username, time: new Date().toLocaleString('ar-SA') }); } io.emit('usersUpdate', users); io.emit('logsUpdate', logs); } res.json({ success: true }); }); app.post('/api/update-cadet', (req, res) => { const { discordId, hours, points, wings, reportTitle, reportContent, editedBy } = req.body; const cadet = cadets.find(c => c.discordId === discordId); if (cadet) { cadet.hours = parseFloat(hours) || 0; cadet.points = parseInt(points) || 0; cadet.wings = wings || []; if (reportTitle && reportContent) { if (!cadet.reports) cadet.reports = []; cadet.reports.push({ title: reportTitle, content: reportContent, date: new Date().toISOString().split('T')[0] }); } logs.unshift({ by: editedBy || 'Admin', action: 'تعديل بيانات ورتبة/ساعات العسكري', target: cadet.name, time: new Date().toLocaleString('ar-SA') }); const sorted = sortCadets(cadets); io.emit('cadetsUpdate', sorted); io.emit('logsUpdate', logs); res.json({ success: true }); } else { res.status(404).json({ error: 'Cadet not found' }); } }); // إعداد الاتصال التفاعلي عبر Socket.IO io.on('connection', (socket) => { socket.emit('cadetsUpdate', sortCadets(cadets)); socket.emit('usersUpdate', users); socket.emit('logsUpdate', logs); }); // فحص عدم النشاط للتحقق من حالة المستخدمين كل 30 ثانية setInterval(() => { const now = Date.now(); let updated = false; users.forEach(u => { if (now - u.lastSeen > 120000 && u.status !== 'no-active') { u.status = 'no-active'; updated = true; } }); if (updated) { io.emit('usersUpdate', users); } }, 30000); const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`Server running on port ${PORT}`); });
